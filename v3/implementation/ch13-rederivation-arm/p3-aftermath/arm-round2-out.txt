    90	    ];
    91	  }
    92	  const problems: string[] = [];
    93	  for (const segment of segments.slice(1)) {
    94	    if (segment === "") {
    95	      problems.push(`${where}: selector path ${JSON.stringify(path)} has an empty segment`);
    96	    } else if (segment.includes("*") && segment !== "*") {
    97	      problems.push(
    98	        `${where}: selector path ${JSON.stringify(path)} uses "*" inside a segment; ` +
    99	          `"*" is legal only as a whole segment`,
   100	      );
   101	    }
   102	  }
   103	  return problems;
   104	}

   106	/** A NORMALIZER operand path: `$`-rooted, dot-separated, with `*` legal as
   107	 * a whole segment (the `.*.` wildcard the hook resolver splits on). */
   108	function normalizerPathProblems(path: string, where: string): string[] {
   109	  const segments = path.split(".");
   110	  if (segments[0] !== "$") {
   111	    return [`${where}: normalizer path ${JSON.stringify(path)} must start with "$"`];
   112	  }
   113	  const problems: string[] = [];
   114	  for (const segment of segments.slice(1)) {
   115	    if (segment === "") {
   116	      problems.push(`${where}: normalizer path ${JSON.stringify(path)} has an empty segment`);
   117	    } else if (segment.includes("*") && segment !== "*") {
   118	      problems.push(
   119	        `${where}: normalizer path ${JSON.stringify(path)} uses "*" inside a segment; ` +
   120	          `"*" is legal only as a whole segment`,
   121	      );
   122	    }
   123	  }
   124	  return problems;
   125	}

   127	/** A FINDING-PATH template is RENDERED, never walked: it may carry closed
   128	 * placeholder slots and must not carry selector syntax. */
   129	function findingPathProblems(path: string, where: string): string[] {
   130	  const problems = slotProblems(path, where);
   131	  const literal = path.replace(/\{\w+\}/gu, "");
   132	  if (literal.startsWith("$") || literal.startsWith("^") || literal.includes("*")) {
   133	    problems.push(
   134	      `${where}: finding path ${JSON.stringify(path)} uses selector syntax; ` +
   135	        `a finding path is rendered, not walked`,
   136	    );
   137	  }
   138	  return problems;
   139	}

   141	/** A reference whose target can only be resolved once the whole tree has
   142	 * been walked. Collected in pass one, resolved in pass two. */
   143	interface PendingRef {
   144	  readonly path: string;
   145	  readonly relation: "keysOf" | "valuesOf" | "collect";
   146	  /** The declaration path of the node that CITED it — the operand of `^`. */
   147	  readonly from: string;
   148	  readonly where: string;
   149	}
   150	
   151	interface Context {
   152	  readonly valueClasses: Readonly<Record<string, NodeDecl>>;
   153	  readonly codes: ReadonlySet<string>;
   154	  readonly tags: string[];
   155	  /** Tags the ENGINE records reliability for — only these can satisfy a
   156	   * `dependsOn` (F4: a substrate or cross-rule tag never gets a status, so
   157	   * depending on one suppresses nothing, forever). */
   158	  readonly nodeTags: Set<string>;
   159	  /** Declaration path → the node at it. The engine addresses positions the
   160	   * same way, so this is what a selector target must resolve against. */
   161	  readonly positions: Map<string, NodeDecl>;
   162	  readonly pending: PendingRef[];
   163	  /** Hook tags whose operand paths are syntactically clean. A path that
   164	   * failed its grammar is NOT resolved a second time — one broken
   165	   * reference must yield one problem, or a maintainer cannot tell how many
   166	   * things are actually wrong. */
   167	  readonly resolvableHooks: Set<string>;
   168	  readonly problems: string[];
   169	}

   171	/** Syntax now; the TARGET's existence and kind are resolved in pass two. */
   172	function selectorProblems(selector: Selector, where: string, from: string, ctx: Context): string[] {
   173	  if ("injected" in selector) return [];
   174	  if ("union" in selector) return selector.union.flatMap((part) => selectorProblems(part, where, from, ctx));
   175	  const relation = "keysOf" in selector ? "keysOf" : "valuesOf" in selector ? "valuesOf" : "collect";
   176	  const path = "keysOf" in selector ? selector.keysOf : "valuesOf" in selector ? selector.valuesOf : selector.collect;
   177	  const syntax = selectorPathProblems(path, where);
   178	  if (syntax.length === 0) ctx.pending.push({ path, relation, from, where });
   179	  return syntax;
   180	}

   182	function checkMembership(
   183	  rule: { readonly target: Selector; readonly message: MessageTemplate; readonly code?: string },
   184	  where: string,
   185	  from: string,
   186	  ctx: Context,
   187	): void {
   188	  ctx.problems.push(...selectorProblems(rule.target, where, from, ctx));
   189	  ctx.problems.push(...slotProblems(rule.message, where));
   190	  if (rule.code !== undefined && !ctx.codes.has(rule.code)) {
   191	    ctx.problems.push(`${where}: issue code ${JSON.stringify(rule.code)} is outside the declared namespace`);
   192	  }
   193	}

   195	/**
   196	 * Walk one node. `channelHonoured` says whether a `channel` mark at THIS
   197	 * position is read by the engine — true only for a field of a `map.fixed`.
   198	 */
   199	function checkNode(
   200	  decl: NodeDecl,
   201	  where: string,
   202	  channelHonoured: boolean,
   203	  ctx: Context,
   204	  siblings: readonly string[] = [],
   205	): void {
   206	  ctx.tags.push(decl.tag);
   207	  ctx.nodeTags.add(decl.tag);
   208	  ctx.positions.set(where, decl);
   209	  if (decl.rows.length === 0) ctx.problems.push(`${where}: node "${decl.tag}" cites no ratified row`);
   210	  if (decl.channel !== undefined && !channelHonoured) {
   211	    ctx.problems.push(
   212	      `${where}: node "${decl.tag}" carries channel ${JSON.stringify(decl.channel)}, ` +
   213	        `which is read only on a field of a map.fixed`,
   214	    );
   215	  }
   216	  if (decl.presence?.code !== undefined && !ctx.codes.has(decl.presence.code)) {
   217	    ctx.problems.push(
   218	      `${where}: issue code ${JSON.stringify(decl.presence.code)} is outside the declared namespace`,
   219	    );
   220	  }
   221	  if (decl.presence?.message !== undefined) ctx.problems.push(...slotProblems(decl.presence.message, where));
   222	
   223	  switch (decl.kind) {
   224	    case "map.fixed": {
   225	      ctx.problems.push(...slotProblems(decl.containerMessage, where), ...slotProblems(decl.unknownMessage, where));
   226	      if (decl.missingMessage !== undefined) ctx.problems.push(...slotProblems(decl.missingMessage, where));
   227	      const fieldNames = Object.keys(decl.fields);
   228	      for (const [name, field] of Object.entries(decl.fields)) {
   229	        checkNode(field, `${where}.${name}`, true, ctx, fieldNames);
   230	      }
   231	      for (const [key, message] of Object.entries(decl.removedKeys ?? {})) {
   232	        ctx.problems.push(...slotProblems(message, `${where}.removedKeys.${key}`));
   233	      }
   234	      return;
   235	    }
   236	    case "map.open": {
   237	      ctx.problems.push(...slotProblems(decl.containerMessage, where));
   238	      if (decl.nonempty !== undefined) ctx.problems.push(...slotProblems(decl.nonempty.message, where));
   239	      if (decl.deepKeyStringness !== undefined) {
   240	        ctx.problems.push(...slotProblems(decl.deepKeyStringness.message, where));
   241	      }
   242	      if (decl.keysSubsetOf !== undefined) checkMembership(decl.keysSubsetOf, `${where}.keysSubsetOf`, where, ctx);
   243	      if (decl.keyClass !== undefined) checkNode(decl.keyClass, `${where}<key>`, false, ctx);
   244	      checkNode(decl.entry, `${where}.*`, false, ctx);
   245	      return;
   246	    }
   247	    case "list": {
   248	      ctx.problems.push(...slotProblems(decl.containerMessage, where));
   249	      if (decl.nonempty !== undefined) ctx.problems.push(...slotProblems(decl.nonempty.message, where));
   250	      if (decl.unique !== undefined) ctx.problems.push(...slotProblems(decl.unique.message, where));
   251	      if (decl.memberOf !== undefined) checkMembership(decl.memberOf, `${where}.memberOf`, where, ctx);
   252	      if (decl.disjointFrom !== undefined) checkMembership(decl.disjointFrom, `${where}.disjointFrom`, where, ctx);
   253	      checkNode(decl.member, `${where}[]`, false, ctx);
   254	      return;
   255	    }
   256	    case "string": {
   257	      if (decl.typeMessage !== undefined) ctx.problems.push(...slotProblems(decl.typeMessage, where));
   258	      if (decl.nonempty !== undefined) ctx.problems.push(...slotProblems(decl.nonempty.message, where));
   259	      if (decl.grammar !== undefined) {
   260	        ctx.problems.push(...slotProblems(decl.grammar.message, where));
   261	        try {
   262	          new RegExp(decl.grammar.re, "u");
   263	        } catch {
   264	          ctx.problems.push(`${where}: grammar ${JSON.stringify(decl.grammar.re)} is not a valid regular expression`);
   265	        }
   266	      }
   267	      if (decl.memberOf !== undefined) checkMembership(decl.memberOf, `${where}.memberOf`, where, ctx);
   268	      return;
   269	    }
   270	    case "integer":
   271	      if (decl.resolvedForm !== undefined) ctx.problems.push(...slotProblems(decl.resolvedForm.message, where));
   272	      return;
   273	    case "enum": {
   274	      ctx.problems.push(...slotProblems(decl.message, where));
   275	      if (decl.code !== undefined && !ctx.codes.has(decl.code)) {
   276	        ctx.problems.push(`${where}: issue code ${JSON.stringify(decl.code)} is outside the declared namespace`);
   277	      }
   278	      return;
   279	    }
   280	    case "union": {
   281	      ctx.problems.push(...slotProblems(decl.message, where));
   282	      for (const [value, message] of Object.entries(decl.removedValues ?? {})) {
   283	        ctx.problems.push(...slotProblems(message, `${where}.removedValues.${value}`));
   284	      }
   285	      if (decl.mapCase !== undefined) checkNode(decl.mapCase, where, false, ctx);
   286	      return;
   287	    }
   288	    case "raw":
   289	      if (decl.containerMessage !== undefined) ctx.problems.push(...slotProblems(decl.containerMessage, where));
   290	      return;
   291	    case "map.plain":
   292	      ctx.problems.push(
   293	        ...slotProblems(decl.containerMessage, where),
   294	        ...slotProblems(decl.canonicalJsonSafe.message, where),
   295	      );
   296	      return;
   297	    case "valueClass": {
   298	      // The measured shame case: an unknown name made the engine return the
   299	      // value unvalidated, so a mistyped class silently switched a rule off.
   300	      if (!(decl.valueClass in ctx.valueClasses)) {
   301	        ctx.problems.push(
   302	          `${where}: value class ${JSON.stringify(decl.valueClass)} is not declared ` +
   303	            `(declared: ${Object.keys(ctx.valueClasses).join(", ")})`,
   304	        );
   305	      }
   306	      return;
   307	    }
   308	    case "delegate": {
   309	      ctx.problems.push(...slotProblems(decl.beltMessage, where));
   310	      // The hand-off reads a SIBLING field to name the registration. A
   311	      // typo there makes the engine read `undefined`, skip the registry
   312	      // and treat the delegation as satisfied.
   313	      if (!siblings.includes(decl.by)) {
   314	        ctx.problems.push(
   315	          `${where}: delegate reads sibling field ${JSON.stringify(decl.by)}, ` +
   316	            `which the enclosing map does not declare (siblings: ${siblings.join(", ")})`,
   317	        );
   318	      }
   319	      return;
   320	    }
   321	  }
   322	}

   324	function checkCrossRule(rule: EqualsRuleDecl, ctx: Context): void {
   325	  const where = `crossRule "${rule.tag}"`;
   326	  ctx.tags.push(rule.tag);
   327	  if (rule.rows.length === 0) ctx.problems.push(`${where}: cites no ratified row`);
   328	  // A cross rule cites from the ROOT: it has no citing node, so `^` has no
   329	  // operand and every path it writes must be document-rooted.
   330	  ctx.problems.push(...selectorProblems(rule.left, `${where}.left`, "$", ctx));
   331	  ctx.problems.push(...selectorProblems(rule.right, `${where}.right`, "$", ctx));
   332	  ctx.problems.push(...findingPathProblems(rule.missingFromLeft.at, `${where}.missingFromLeft.at`));
   333	  ctx.problems.push(...findingPathProblems(rule.missingFromRight.at, `${where}.missingFromRight.at`));
   334	  ctx.problems.push(...slotProblems(rule.missingFromLeft.message, `${where}.missingFromLeft`));
   335	  ctx.problems.push(...slotProblems(rule.missingFromRight.message, `${where}.missingFromRight`));
   336	}

   338	function checkHook(hook: NormalizerHookDecl, ctx: Context): void {
   339	  const where = `normalizer "${hook.tag}"`;
   340	  ctx.tags.push(hook.tag);
   341	  if (hook.rows.length === 0) ctx.problems.push(`${where}: cites no ratified row`);
   342	  const pathProblems = [...normalizerPathProblems(hook.over, `${where}.over`)];
   343	  if (hook.hook === "expandAdvancesRound") {
   344	    pathProblems.push(...normalizerPathProblems(hook.advanceSet, `${where}.advanceSet`));
   345	  }
   346	  ctx.problems.push(...pathProblems);
   347	  if (pathProblems.length === 0) ctx.resolvableHooks.add(hook.tag);
   348	}

   350	/** The node a declaration path addresses, following a value-class
   351	 * reference to the class it names — the kind a selector actually meets. */
   352	function nodeAt(path: string, ctx: Context): NodeDecl | undefined {
   353	  const node = ctx.positions.get(path);
   354	  if (node?.kind !== "valueClass") return node;
   355	  return ctx.valueClasses[node.valueClass];
   356	}

   358	/** Pass two for selectors: does the target POSITION exist, and is its kind
   359	 * one the relation can read? A syntactically perfect path to a node that
   360	 * does not exist made the rule vanish without a word. */
   361	function resolveSelectors(ctx: Context): void {
   362	  for (const ref of ctx.pending) {
   363	    const segments = ref.path.split(".");
   364	    const head = segments.shift();
   365	    const base = head === "$" ? "$" : ref.from.split(".").slice(0, -1).join(".");
   366	    const target = segments.length === 0 ? base : `${base}.${segments.join(".")}`;
   367	    const node = nodeAt(target, ctx);
   368	    if (node === undefined) {
   369	      ctx.problems.push(
   370	        `${ref.where}: selector path ${JSON.stringify(ref.path)} addresses ${JSON.stringify(target)}, ` +
   371	          `which the declaration does not declare`,
   372	      );
   373	      continue;
   374	    }
   375	    const wanted =
   376	      ref.relation === "keysOf"
   377	        ? ["map.open", "map.fixed", "map.plain"]
   378	        : ref.relation === "valuesOf"
   379	          ? ["list"]
   380	          : ["string", "integer", "enum", "raw"];
   381	    if (!wanted.includes(node.kind)) {
   382	      ctx.problems.push(
   383	        `${ref.where}: ${ref.relation}(${JSON.stringify(ref.path)}) addresses a ${node.kind}; ` +
   384	          `${ref.relation} reads ${wanted.join(" | ")}`,
   385	      );
   386	    }
   387	  }
   388	}

   390	/** Pass two for the normalizer: its operand paths address declared
   391	 * positions, and its dynamic FIELD names are fields of the node it lands
   392	 * on. An unresolved `edges` produced an empty flag map in silence. */
   393	function resolveHooks(surface: SurfaceDecl, ctx: Context): void {
   394	  for (const hook of surface.normalizers) {
   395	    if (!ctx.resolvableHooks.has(hook.tag)) continue;
   396	    const where = `normalizer "${hook.tag}"`;
   397	    const over = ctx.positions.get(hook.over);
   398	    if (over === undefined) {
   399	      ctx.problems.push(`${where}.over: ${JSON.stringify(hook.over)} is not a declared position`);
   400	      continue;
   401	    }
   402	    if (over.kind !== "map.open") {
   403	      ctx.problems.push(`${where}.over: ${JSON.stringify(hook.over)} is a ${over.kind}; the hook walks a map.open`);
   404	      continue;
   405	    }
   406	    const fieldsOf = (path: string): { readonly node: NodeDecl | undefined; readonly names: readonly string[] } => {
   407	      const node = ctx.positions.get(path);
   408	      return { node, names: node?.kind === "map.fixed" ? Object.keys(node.fields) : [] };
   409	    };
   410	    if (hook.hook === "expandAdvancesRound") {
   411	      if (ctx.positions.get(hook.advanceSet) === undefined) {
   412	        ctx.problems.push(
   413	          `${where}.advanceSet: ${JSON.stringify(hook.advanceSet)} is not a declared position`,
   414	        );
   415	      }
   416	      const entry = fieldsOf(`${hook.over}.*`);
   417	      for (const [label, name] of [["edges", hook.edges], ["into", hook.into]] as const) {
   418	        if (!entry.names.includes(name)) {
   419	          ctx.problems.push(
   420	            `${where}.${label}: ${JSON.stringify(name)} is not a field of ${JSON.stringify(`${hook.over}.*`)} ` +
   421	              `(fields: ${entry.names.join(", ")})`,
   422	          );
   423	        }
   424	      }
   425	      continue;
   426	    }
   427	    const member = fieldsOf(`${hook.over}.*[]`);
   428	    for (const name of [...hook.carry, hook.into]) {
   429	      if (!member.names.includes(name)) {
   430	        ctx.problems.push(
   431	          `${where}: ${JSON.stringify(name)} is not a field of ${JSON.stringify(`${hook.over}.*[]`)} ` +
   432	            `(fields: ${member.names.join(", ")})`,
   433	        );
   434	      }
   435	    }
   436	  }
   437	}

   439	function checkBranch(
   440	  branch: SubstrateBranch,
   441	  where: string,
   442	  ctx: Context,
   443	  messages: readonly MessageTemplate[] = [],
   444	): void {
   445	  ctx.tags.push(branch.tag);
   446	  if (branch.rows.length === 0) ctx.problems.push(`${where}: cites no ratified row`);
   447	  for (const message of messages) ctx.problems.push(...slotProblems(message, where));
   448	}

   450	/** Every `dependsOn` tag any rule names, with the site that named it. */
   451	function collectDependsOn(decl: NodeDecl, where: string, into: { tag: string; where: string }[]): void {
   452	  const add = (tags: readonly string[] | undefined, site: string): void => {
   453	    for (const tag of tags ?? []) into.push({ tag, where: site });
   454	  };
   455	  switch (decl.kind) {
   456	    case "map.fixed":
   457	      for (const [name, field] of Object.entries(decl.fields)) collectDependsOn(field, `${where}.${name}`, into);
   458	      return;
   459	    case "map.open":
   460	      add(decl.keysSubsetOf?.dependsOn, `${where}.keysSubsetOf`);
   461	      if (decl.keyClass !== undefined) collectDependsOn(decl.keyClass, `${where}<key>`, into);
   462	      collectDependsOn(decl.entry, `${where}.*`, into);
   463	      return;
   464	    case "list":
   465	      add(decl.memberOf?.dependsOn, `${where}.memberOf`);
   466	      add(decl.disjointFrom?.dependsOn, `${where}.disjointFrom`);
   467	      collectDependsOn(decl.member, `${where}[]`, into);
   468	      return;
   469	    case "string":
   470	      add(decl.memberOf?.dependsOn, `${where}.memberOf`);
   471	      return;
   472	    case "union":
   473	      if (decl.mapCase !== undefined) collectDependsOn(decl.mapCase, where, into);
   474	      return;
   475	    case "delegate":
   476	      add(decl.dependsOn, where);
   477	      return;
   478	    default:
   479	      return;
   480	  }
   481	}

   483	/**
   484	 * Every unresolved reference in a declaration, as a list of problems.
   485	 * Exported so the guard's own fixtures can assert WHICH reference failed
   486	 * rather than only that something did.
   487	 */
   488	export function closureProblems(surface: SurfaceDecl): readonly string[] {
   489	  const ctx: Context = {
   490	    valueClasses: surface.valueClasses,
   491	    codes: new Set(surface.substrate.codes.values),
   492	    tags: [],
   493	    nodeTags: new Set(),
   494	    positions: new Map(),
   495	    pending: [],
   496	    resolvableHooks: new Set(),
   497	    problems: [],
   498	  };
   499	
   500	  const substrate = surface.substrate;
   501	  checkBranch(substrate.read, "substrate.read", ctx, [substrate.read.message]);
   502	  checkBranch(substrate.parse.directive, "substrate.parse.directive", ctx, [substrate.parse.directive.message]);
   503	  checkBranch(substrate.parse.duplicateKeys, "substrate.parse.duplicateKeys", ctx, [
   504	    substrate.parse.duplicateKeys.message,
   505	  ]);
   506	  checkBranch(substrate.resolve.graph, "substrate.resolve.graph", ctx, [substrate.resolve.graph.message]);
   507	  checkBranch(substrate.codes, "substrate.codes", ctx);
   508	  checkBranch(substrate.internalFailure, "substrate.internalFailure", ctx, [substrate.internalFailure.message]);
   509	
   510	  checkNode(surface.root, "$", false, ctx);
   511	  for (const [name, decl] of Object.entries(surface.valueClasses)) {
   512	    checkNode(decl, `valueClass "${name}"`, false, ctx);
   513	    // A value-class definition's tag is never STATUSED by the engine (the
   514	    // reference's tag is), so it must not count as a dependsOn target.
   515	    ctx.nodeTags.delete(decl.tag);
   516	  }
   517	  for (const rule of surface.crossRules) checkCrossRule(rule, ctx);
   518	  for (const hook of surface.normalizers) checkHook(hook, ctx);
   519	
   520	  // Tags must be unique, or a contract pointer is ambiguous.
   521	  const seen = new Set<string>();
   522	  for (const tag of ctx.tags) {
   523	    if (seen.has(tag)) ctx.problems.push(`declaration tag ${JSON.stringify(tag)} is declared more than once`);
   524	    seen.add(tag);
   525	  }
   526	
   527	  // `dependsOn` names a tag the ENGINE gives a reliability status to.
   528	  // Any other declared tag resolves by name and suppresses nothing.
   529	  const dependencies: { tag: string; where: string }[] = [];
   530	  collectDependsOn(surface.root, "$", dependencies);
   531	  for (const [name, decl] of Object.entries(surface.valueClasses)) {
   532	    collectDependsOn(decl, `valueClass "${name}"`, dependencies);
   533	  }
   534	  for (const rule of surface.crossRules) {
   535	    for (const tag of rule.dependsOn ?? []) dependencies.push({ tag, where: `crossRule "${rule.tag}"` });
   536	  }
   537	  for (const dependency of dependencies) {
   538	    if (ctx.nodeTags.has(dependency.tag)) continue;
   539	    ctx.problems.push(
   540	      seen.has(dependency.tag)
   541	        ? `${dependency.where}: dependsOn names ${JSON.stringify(dependency.tag)}, which is declared but is ` +
   542	          `not a node the engine records a status for — depending on it suppresses nothing`
   543	        : `${dependency.where}: dependsOn names ${JSON.stringify(dependency.tag)}, which is not a declared tag`,
   544	    );
   545	  }
   546	
   547	  resolveSelectors(ctx);
   548	  resolveHooks(surface, ctx);
   549	  return ctx.problems;
   550	}

   552	function deepFreeze<T>(value: T): T {
   553	  if (typeof value !== "object" || value === null || Object.isFrozen(value)) return value;
   554	  Object.freeze(value);
   555	  for (const key of Object.getOwnPropertyNames(value)) {
   556	    deepFreeze((value as Record<string, unknown>)[key]);
   557	  }
   558	  return value;
   559	}

   561	/**
   562	 * Make a surface declaration usable: CLOSE it, then FREEZE it. A
   563	 * declaration carrying an unresolved reference never becomes a surface —
   564	 * the throw happens at module load, before any document is validated.

   317	/** Walk a declaration path over the value graph, expanding `*` over open
   318	 * maps. `..` resolves one level up from the citing frame. A prefix that
   319	 * has not been evaluated yet answers PENDING, not "unreliable". */
   320	function resolvePath(
   321	  run: Run,
   322	  from: Frame,
   323	  path: DeclPath,
   324	): { readonly status: "ok"; readonly nodes: unknown[] } | { readonly status: "unreliable" | "pending" } {
   325	  const segments = path.split(".");
   326	  let nodes: unknown[];
   327	  let declPrefix: string;
   328	  const head = segments.shift();
   329	  if (head === "$") {
   330	    nodes = [run.root];
   331	    declPrefix = "$";
   332	  } else if (head === "^") {
   333	    // Parent-relative: the citing node's own container.
   334	    nodes = [from.parentValue];
   335	    declPrefix = from.decl.split(".").slice(0, -1).join(".");
   336	  } else {
   337	    return UNRELIABLE;
   338	  }
   339	  for (const segment of segments) {
   340	    declPrefix = `${declPrefix}.${segment}`;
   341	    if (!run.reliable(declPrefix)) return UNRELIABLE;
   342	    if (!run.completed.has(declPrefix)) return PENDING;
   343	    const next: unknown[] = [];
   344	    for (const node of nodes) {
   345	      if (!isContainer(node)) return UNRELIABLE;
   346	      if (segment === "*") {
   347	        for (const [key, child] of entriesOf(node)) {
   348	          if (typeof key === "string") next.push(child);
   349	        }
   350	      } else {
   351	        if (!containerHas(node, segment)) return UNRELIABLE;
   352	        next.push(containerGet(node, segment));
   353	      }
   354	    }
   355	    nodes = next;
   356	  }
   357	  return { status: "ok", nodes };
   358	}

    50	 * COMPLETE per-edge boolean map against the declared advancing set. An
    51	 * absent declaration yields all-false (ch11-C38's none-default); the

    55	function expandFlagMaps(
    56	  root: Record<string, unknown>,
    57	  hook: Extract<NormalizerHookDecl, { hook: "expandAdvancesRound" }>,
    58	): void {
    59	  const entries = at(root, hook.over);
    60	  if (!isRecord(entries)) return;
    61	  const declared = at(root, hook.advanceSet);
    62	  const advancing = new Set<string>(Array.isArray(declared) ? (declared as string[]) : []);
    63	  for (const key of Object.keys(entries)) {
    64	    const entry = ownGet(entries, key);
    65	    if (!isRecord(entry)) continue;
    66	    const edges = ownGet(entry, hook.edges);
    67	    const flags: Record<string, boolean> = {};
    68	    if (isRecord(edges)) {
    69	      for (const edge of Object.keys(edges)) {
    70	        defineOwn(flags, edge, advancing.has(ownGet(edges, edge) as string));
    71	      }
    72	    }

    74	  }

    78	 * R4's second member (ch11-C20/C5): write each registration's EFFECTIVE
    79	 * config into the binding's single config surface. The binding is rebuilt
    80	 * from its CARRIED fields plus the effective config — the one downstream
    81	 * config form.
    82	 */
    83	function materializeConfigs(
    84	  root: Record<string, unknown>,
    85	  hook: Extract<NormalizerHookDecl, { hook: "materializeEffectiveConfigs" }>,
    86	  effectiveConfigs: ReadonlyMap<string, unknown>,
    87	): void {
    88	  // `over` addresses the pipeline map through two open-map levels:
    89	  // `$.steps.*.gates` — the entry key and the event key are the wildcards.
    90	  const [ownerPath, ...rest] = hook.over.split(".*.");
    91	  const owners = at(root, ownerPath ?? "$");
    92	  if (!isRecord(owners)) return;
    93	  const inner = rest.join(".*.");

    95	    const owner = ownGet(owners, ownerKey);
    96	    if (!isRecord(owner)) continue;

    98	    if (!isRecord(pipelines)) continue;
    99	    const rebuiltPipelines: Record<string, unknown> = {};
   100	    for (const eventKey of Object.keys(pipelines)) {
   101	      const pipeline = ownGet(pipelines, eventKey);
   102	      if (!Array.isArray(pipeline)) continue;
   103	      const base = `${ownerPath === "$" ? "" : `${(ownerPath ?? "").replace("$.", "")}.`}${ownerKey}.${inner}.${eventKey}`;
   104	      defineOwn(
   105	        rebuiltPipelines,
   106	        eventKey,
   107	        pipeline.map((binding: unknown, index: number) => {
   108	          const carried: Record<string, unknown> = {};
   109	          if (isRecord(binding)) {
   110	            for (const field of hook.carry) defineOwn(carried, field, ownGet(binding, field));
   111	          }
   112	          defineOwn(carried, hook.into, effectiveConfigs.get(`${base}[${String(index)}]`));
   113	          return carried;
   114	        }),
   115	      );
   116	    }
   117	    defineOwn(owner, inner, rebuiltPipelines);
   118	  }

    97	  // Map still merges them). YAML canonical int equality also makes
    98	  // 0 == -0; NaN == NaN holds under both forms.
    99	  if (isScalar(a) && isScalar(b)) return sameValueZero(a.value, b.value);
   100	  if (isAlias(a) && isAlias(b)) return a.source === b.source;
   101	  if (isAlias(a) && nodeAnchor(b) !== undefined) return nodeAnchor(b) === a.source;
   102	  if (nodeAnchor(a) !== undefined && isAlias(b)) return nodeAnchor(a) === b.source;
   103	  if (isPair(a) && isPair(b)) {
   104	    return yamlNodeEqual(a.key, b.key, doc, compared) && yamlNodeEqual(a.value, b.value, doc, compared);
   105	  }
   106	  if (isSeq(a) && isSeq(b)) {
   107	    return a.items.length === b.items.length
   108	      && a.items.every((item, index) => yamlNodeEqual(item, b.items[index], doc, compared));
   109	  }
   110	  if (isMap(a) && isMap(b)) {
   111	    if (a.items.length !== b.items.length) return false;
   112	    const matched = new Set<number>();
   113	    return a.items.every((pair) => {
   114	      const index = b.items.findIndex((candidate, candidateIndex) => (
   115	        !matched.has(candidateIndex) && yamlNodeEqual(pair, candidate, doc, compared)
   116	      ));
   117	      if (index < 0) return false;
   118	      matched.add(index);
   119	      return true;
   120	    });
   121	  }
   122	  return false;
   123	}

   130	function resolvedDuplicateFindings(doc: Document, lineCounter: LineCounter): readonly PositionedFinding[] {
   131	  const findings: PositionedFinding[] = [];
   132	  const visit = (node: unknown): void => {
   133	    if (isMap(node)) {
   134	      for (let right = 1; right < node.items.length; right += 1) {
   135	        const rightKey = node.items[right]?.key;
   136	        const earlierKeys = node.items.slice(0, right).map((pair) => pair.key);
   137	        const hasComposeTimeMatch = earlierKeys.some((leftKey) => yamlNodeEqual(leftKey, rightKey));
   138	        const hasResolvedMatch = !hasComposeTimeMatch
   139	          && earlierKeys.some((leftKey) => yamlNodeEqual(leftKey, rightKey, doc));
   140	        if (!hasResolvedMatch) continue;
   141	
   142	        const range = (rightKey as { readonly range?: readonly number[] } | null)?.range;
   143	        const offset = range?.[0] ?? Number.MAX_SAFE_INTEGER;
   144	        const pos = Number.isSafeInteger(offset) ? lineCounter.linePos(offset) : undefined;
   145	        findings.push({
   146	          offset,
   147	          finding: {
   148	            stage: "parse",
   149	            ...(pos && pos.line > 0 ? { line: pos.line, col: pos.col } : {}),
   150	            message: SUBSTRATE.parse.duplicateKeys.message,
   151	          },
   152	        });
   153	      }
   154	      for (const pair of node.items) visit(pair);
   155	    } else if (isSeq(node)) {
   156	      for (const item of node.items) visit(item);
   157	    } else if (isPair(node)) {
   158	      visit(node.key);
   159	      visit(node.value);
   160	    }
   161	  };
   162	  visit(doc.contents);
   163	  return findings;

   177	  // READ stage, decode half (G3/draft C6): strict UTF-8 — invalid bytes
   178	  // reject with a content-free message, never the silent U+FFFD repair.

   183	    return fail("read", [

   185	        stage: "read",
   186	        ...(opts.path !== undefined ? { path: opts.path } : {}),
   187	        message: SUBSTRATE.read.message,
   188	      },
   189	    ]);
   190	  }
   191	
   192	  // PARSE stage (G4–G7/draft C2–C4, C34): the document API with BOTH
   193	  // errors AND warnings promoted (fail-closed), plus the directive

   196	  const lineCounter = new LineCounter();

   198	    doc = parseDocument(text, {
   199	      lineCounter,
   200	      uniqueKeys: (a: ParsedNode, b: ParsedNode) => yamlNodeEqual(a, b),
   201	    });

   204	    // doc.errors — the stage catch is the C22 every-stage belt.
   205	    return fail("parse", [{ stage: "parse", message: errorMessage(error) }]);

   211	    // zero warnings, the full 1.1 trap restored). Synthesized: the

   213	    parseFindings.push({
   214	      stage: "parse",
   215	      message: renderMessage(SUBSTRATE.parse.directive.message, { key: yamlDirective.version }),
   216	    });
   217	  }
   218	  const positionedErrors: PositionedFinding[] = [
   219	    ...doc.errors.map((error) => ({ offset: error.pos[0], finding: parseFinding(error) })),
   220	    ...resolvedDuplicateFindings(doc, lineCounter),
   221	  ].sort((a, b) => a.offset - b.offset);
   222	  for (const { finding } of positionedErrors) {
   223	    parseFindings.push(finding);
   224	  }
   225	  for (const warning of doc.warnings) {
   226	    parseFindings.push(parseFinding(warning));
   227	  }
   228	  if (parseFindings.length > 0) {
   229	    return fail("parse", parseFindings);
   230	  }

   232	  // RESOLVE stage (G8/draft C5+C36): toJS — the alias-amplification
   233	  // guard throws HERE, invisible to doc.errors (probes P8/P14/P16).
   234	  let value: unknown;
   235	  try {

   238	    // `1` and `"1"` before V5 can reject the non-string key.

   241	    return fail("resolve", [{ stage: "resolve", message: errorMessage(error) }]);
   242	  }
   243	
   244	  // VALIDATE stage: ONE run of the template surface's declared schema
   245	  // (ADR-019 D1) over the resolved value graph, on the FILE channel so
   246	  // the source-bearing lanes have an operand. The two rungs this stage
   247	  // used to have — the source-form walk and the admission rung — are one
   248	  // computation over one declaration; their findings were already
   249	  // required to accumulate in ONE `validate` result (ch11-P4 F7), and
   250	  // now they cannot fail to. The result stays XOR (ch8-C22): an admitted
   251	  // template escapes only when the run is findings-free.

   253	    const admitted = admitFromSource(value, doc, text, opts.catalog ?? EMPTY_CATALOG);
   254	    if (!admitted.ok) {
   255	      return fail("validate", admitted.findings);
   256	    }
   257	    return { ok: true, template: admitted.template };
   258	  } catch (error) {
   259	    // The C22 every-stage belt: no known input reaches this (V15's
   260	    // cycle rule is a FINDING, not a throw), but the stage still maps.
   261	    return fail("validate", [
   262	      {
   263	        path: SUBSTRATE.internalFailure.path,
   264	        message: renderMessage(SUBSTRATE.internalFailure.message, { value: errorMessage(error) }),
   265	      },
   266	    ]);
   267	  }

   317	/** Walk a declaration path over the value graph, expanding `*` over open
   318	 * maps. `..` resolves one level up from the citing frame. A prefix that
   319	 * has not been evaluated yet answers PENDING, not "unreliable". */
   320	function resolvePath(
   321	  run: Run,
   322	  from: Frame,
   323	  path: DeclPath,
   324	): { readonly status: "ok"; readonly nodes: unknown[] } | { readonly status: "unreliable" | "pending" } {
   325	  const segments = path.split(".");
   326	  let nodes: unknown[];
   327	  let declPrefix: string;
   328	  const head = segments.shift();
   329	  if (head === "$") {
   330	    nodes = [run.root];
   331	    declPrefix = "$";
   332	  } else if (head === "^") {
   333	    // Parent-relative: the citing node's own container.
   334	    nodes = [from.parentValue];
   335	    declPrefix = from.decl.split(".").slice(0, -1).join(".");
   336	  } else {
   337	    return UNRELIABLE;
   338	  }
   339	  for (const segment of segments) {
   340	    declPrefix = `${declPrefix}.${segment}`;
   341	    if (!run.reliable(declPrefix)) return UNRELIABLE;
   342	    if (!run.completed.has(declPrefix)) return PENDING;
   343	    const next: unknown[] = [];
   344	    for (const node of nodes) {
   345	      if (!isContainer(node)) return UNRELIABLE;
   346	      if (segment === "*") {
   347	        for (const [key, child] of entriesOf(node)) {
   348	          if (typeof key === "string") next.push(child);
   349	        }
   350	      } else {
   351	        if (!containerHas(node, segment)) return UNRELIABLE;
   352	        next.push(containerGet(node, segment));
   353	      }
   354	    }
   355	    nodes = next;
   356	  }
   357	  return { status: "ok", nodes };
   358	}

   550	  switch (decl.kind) {

   591	function evalMapFixed(
   592	  run: Run,
   593	  decl: Extract<NodeDecl, { kind: "map.fixed" }>,
   594	  frame: Frame,
   595	): NodeResult {
   596	  const value = frame.value;
   597	  const slots = baseSlots(frame);
   598	  if (!isContainer(value)) {
   599	    run.emit(frame.path, render(decl.containerMessage, slots));
   600	    return { ok: false };
   601	  }
   602	  const container = value;
   603	  const fieldNames = Object.keys(decl.fields);
   604	  // A field declared for ONE channel is legal only there — the same
   605	  // channel scoping D1 applies to source-bearing attributes, applied to
   606	  // keyset membership.
   607	  const inChannel = (name: string): boolean => {
   608	    const channel = decl.fields[name]?.channel;
   609	    return channel === undefined || channel === "both" || channel === run.channel.kind;
   610	  };
   611	  // A channel-scoped field is scoped WHOLE: out of channel it is neither a
   612	  // legal key, nor a missing one, nor evaluated (design review F7).
   613	  const scoped = fieldNames.filter(inChannel);
   614	  const legal = new Set(scoped);
   615	  const normalized: Record<string, unknown> = {};
   616	  const local: LocalOk = new Map();
   617	
   618	  const emitUnknownKeys = (): void => {
   619	    for (const [key] of entriesOf(container)) {
   620	      if (typeof key !== "string") {
   621	        run.emit(
   622	          frame.path,
   623	          render(decl.unknownMessage, { ...slots, key, value: key, keys: scoped.join(", ") }),
   624	        );
   625	        continue;
   626	      }
   627	      if (legal.has(key)) continue;
   628	      const at = joinPath(frame.path, key);
   629	      const removed = decl.removedKeys?.[key];
   630	      const keySlots: Slots = { ...slots, path: at, key, value: key, keys: scoped.join(", ") };
   631	      run.emit(at, render(removed ?? decl.unknownMessage, keySlots));
   632	    }
   633	  };
   634	
   635	  const emitMissing = (name: string): void => {
   636	    const field = decl.fields[name];
   637	    const presence = field?.presence;
   638	    if (field === undefined || presence?.required !== true) return;
   639	    if (containerHas(container, name)) return;
   640	    if (presence.foldedIntoTypeLane === true) {
   641	      // Absence is this node's OWN type lane (a binding's `uses`).
   642	      evaluateNode(run, field, childFrame(frame, name, undefined, container), local);
   643	      local.set(field.tag, false);
   644	      return;
   645	    }
   646	    const at = presence.at === "self" ? joinPath(frame.path, name) : frame.path;
   647	    run.emit(
   648	      at,
   649	      render(presence.message ?? decl.missingMessage ?? DEFAULT_MISSING, { ...slots, path: at, key: name }),
   650	      presence.code,
   651	    );
   652	    local.set(field.tag, false);
   653	  };
   654	
   655	  const evalOneField = (name: string): void => {
   656	    const field = decl.fields[name];
   657	    if (field === undefined) return;
   658	    if (!containerHas(container, name)) {
   659	      // An ABSENT field is where a declared `default:` materializes — the
   660	      // schema-side half of ADR-019 D3.
   661	      if (field.default !== undefined) defineOwn(normalized, name, field.default);
   662	      return;
   663	    }
   664	    const child = childFrame(frame, name, containerGet(container, name), container);
   665	    const result = evaluateNode(run, field, child, local);
   666	    local.set(field.tag, result.ok);
   667	    if (result.ok && result.value !== undefined) defineOwn(normalized, name, result.value);
   668	  };
   669	
   670	  const order = decl.laneOrder ?? "unknownThenPerKey";
   671	  if (order === "missingThenUnknown") {
   672	    for (const name of scoped) emitMissing(name);
   673	    emitUnknownKeys();
   674	    for (const name of scoped) evalOneField(name);
   675	  } else if (order === "unknownThenMissingThenValues") {
   676	    emitUnknownKeys();
   677	    for (const name of scoped) emitMissing(name);
   678	    for (const name of scoped) evalOneField(name);
   679	  } else {
   680	    emitUnknownKeys();
   681	    for (const name of scoped) {
   682	      if (!containerHas(container, name)) {
   683	        emitMissing(name);
   684	        evalOneField(name);
   685	        continue;
   686	      }
   687	      evalOneField(name);
   688	    }
   689	  }
   690	  return { ok: true, value: normalized };
   691	}

   321	  run: Run,

   323	  path: DeclPath,

   360	function evaluateSelector(run: Run, from: Frame, selector: Selector): SelectorResult {
   361	  if ("injected" in selector) {
   362	    // D8's ADMITTED widening: the selector root is an INJECTED set. A
   363	    // registry resolves rather than enumerates, so membership is asked of
   364	    // it directly (`isMember`); the set itself is never listed.
   365	    return run.registry(selector.injected) === undefined ? UNRELIABLE : { status: "ok", values: [] };
   366	  }
   367	  if ("union" in selector) {
   368	    const parts = selector.union.map((part) => evaluateSelector(run, from, part));
   369	    if (parts.some((part) => part.status === "pending")) return PENDING;
   370	    if (parts.some((part) => part.status === "unreliable")) return UNRELIABLE;
   371	    return {
   372	      status: "ok",
   373	      values: parts.flatMap((part) => (part.status === "ok" ? part.values : [])),
   374	    };
   375	  }
   376	  const target = "keysOf" in selector ? selector.keysOf : "valuesOf" in selector ? selector.valuesOf : selector.collect;
   377	  const resolved = resolvePath(run, from, target);
   378	  if (resolved.status !== "ok") return resolved;
   379	  const values: string[] = [];
   380	  for (const node of resolved.nodes) {
   381	    if ("keysOf" in selector) {
   382	      if (!isContainer(node)) return UNRELIABLE;
   383	      for (const [key] of entriesOf(node)) if (typeof key === "string") values.push(key);
   384	    } else if ("valuesOf" in selector) {
   385	      if (!Array.isArray(node)) return UNRELIABLE;
   386	      for (const item of node) if (typeof item === "string") values.push(item);
   387	    } else {
   388	      if (typeof node !== "string") return UNRELIABLE;
   389	      values.push(node);
   390	    }
   391	  }
   392	  return { status: "ok", values };
   393	}

   403	  run: Run,
   404	  frame: Frame,
   405	  rule: MembershipRule,
