#!/usr/bin/env bash
# M3 — the admitted-VALUE re-pin floor (packet row D14). EXECUTABLE RECEIPT.
#
# Rewritten 2026-08-09 after arm gate 1 found the first version to be a
# narrated command outline: its three essential mutations were COMMENTS and
# its symlink target was a `<repo>` placeholder, so no third party could
# re-run it. This script is the whole measurement — it builds both trees,
# applies the growth by exact anchored patch, runs both suites, diffs the
# failure sets, asserts the floor, and deletes its throwaway trees.
#
# WHAT IS MEASURED: which STANDING assertions go red once every admitted
# value carries the produced ref positions. The MECHANISM of the fill is
# deliberately not the subject — a declared hook and the direct patch below
# produce the same admitted value, and the red set is a function of that
# value, not of how it was produced. The declared-hook realization is D4's
# and is verified separately.
#
# Every patch ASSERTS its own anchor hit count, so a future edit that moves
# an anchor fails the receipt loudly instead of measuring the wrong tree.
set -euo pipefail

SRC="$(git rev-parse --show-toplevel)"
BASE="$(mktemp -d)"; GROW="$(mktemp -d)"
trap 'rm -rf "$BASE" "$GROW"' EXIT

# The v3 package is its own vitest root (`pnpm --dir v3 test`), so each
# throwaway tree IS a v3 root — not a repo root with v3 inside it.
for T in "$BASE" "$GROW"; do
  rsync -a --exclude node_modules "$SRC/v3/" "$T/"
  ln -s "$SRC/v3/node_modules" "$T/node_modules"
done

python3 - "$GROW" <<'PY'
import sys, pathlib
g = pathlib.Path(sys.argv[1])

# (1) The gate position's declared empty-list default.
tf = g / "src/definition/schema/templateFormat.ts"
t = tf.read_text()
a1 = """    contextBlockRefs: {
      kind: "valueClass",
      tag: "d-ctx-gate-refs",
      rows: ["ch13v2-C6"],
      valueClass: "blockIdList",
      label: "contextBlockRefs",
    },"""
assert t.count(a1) == 1, f"[d-ctx-gate-refs] anchor hits: {t.count(a1)}"
t = t.replace(a1, a1.replace('      label: "contextBlockRefs",',
                             '      label: "contextBlockRefs",\n      default: [],'))

# (2) The effective-config carry gains the gate-position field.
a2 = '      carry: ["uses"],'
assert t.count(a2) == 1, f"carry anchor hits: {t.count(a2)}"
t = t.replace(a2, '      carry: ["uses", "contextBlockRefs"],')
tf.write_text(t)

# (3) The nested-source fill at both config positions, applied directly.
#     The produced field is an ENTRY-LEVEL sibling (D3: the two new field\n#     nodes join [d-roles-entry] and [d-step]), sourced from the nested\n#     config. This stands in for the declared hook; see the header.
nz = g / "src/definition/schema/normalizer.ts"
n = nz.read_text()
a3 = """  if (!isRecord(value)) return value;
  for (const hook of surface.normalizers) {"""
assert n.count(a3) == 1, f"normalize() anchor hits: {n.count(a3)}"
fill = """  if (!isRecord(value)) return value;
  for (const parentKey of ["steps", "roles"]) {
    const parent = (value as Record<string, unknown>)[parentKey];
    if (!isRecord(parent)) continue;
    const configKey = parentKey === "steps" ? "agentConfig" : "defaultAgentConfig";
    for (const entry of Object.values(parent)) {
      if (!isRecord(entry)) continue;
      const cfg = (entry as Record<string, unknown>)[configKey];
      const authored = isRecord(cfg) ? (cfg as Record<string, unknown>)["promptConcernRefs"] : undefined;
      (entry as Record<string, unknown>)["promptConcernRefs"] = Array.isArray(authored) ? authored : [];
    }
  }
  for (const hook of surface.normalizers) {"""
n = n.replace(a3, fill)
nz.write_text(n)
print("growth applied: 2 declaration patches + 1 normalizer fill")
PY

run() { (cd "$1" && npx vitest run src/definition src/drift 2>&1) || true; }
run "$BASE" > "$BASE/out.txt"
run "$GROW" > "$GROW/out.txt"

# A clean tree yields no matches, and an empty grep exits 1 — which under
# `set -euo pipefail` would kill the run at exactly the moment the
# measurement succeeded. The `|| true` is load-bearing, not defensive noise.
# Two shapes matter here. A clean tree yields no matches and an empty grep
# exits 1, which under `set -euo pipefail` would kill the run at exactly the
# moment the measurement succeeded — hence `|| true`. And vitest prints each
# failure TWICE, once as `file > suite > name` and once as the bare name in
# its summary; keeping only the qualified form is what makes the count the
# number of failing ASSERTIONS rather than twice that.
fails() { { grep -oE '^ *(×|FAIL|✗) .*' "$1" || true; } | sed -E 's/^ *(×|FAIL|✗) //; s/ [0-9]+ms$//' | grep ' > ' | sort -u; }
{ fails "$BASE/out.txt" || true; } > "$BASE/fails.txt"
{ fails "$GROW/out.txt" || true; } > "$GROW/fails.txt"

echo "=== baseline summary ==="; grep -E '^ *Tests +' "$BASE/out.txt" || true
echo "=== growth summary ===";   grep -E '^ *Tests +' "$GROW/out.txt" || true
echo
echo "=== GROWTH-ONLY failures (the floor) ==="
comm -13 "$BASE/fails.txt" "$GROW/fails.txt" | tee "$GROW/only.txt"
echo
echo "=== BASELINE-ONLY failures (must be empty) ==="
comm -23 "$BASE/fails.txt" "$GROW/fails.txt"

N="$(wc -l < "$GROW/only.txt" | tr -d ' ')"
B="$(comm -23 "$BASE/fails.txt" "$GROW/fails.txt" | wc -l | tr -d ' ')"
[ "$B" = "0" ] || { echo "RECEIPT FAILED: baseline-only failures exist"; exit 1; }
FILES="$(sed -E 's/ >.*//' "$GROW/only.txt" | sort -u | wc -l | tr -d ' ')"
[ "$N" -ge 8 ] || { echo "RECEIPT FAILED: floor is $N, expected at least 8"; exit 1; }
[ "$FILES" -ge 4 ] || { echo "RECEIPT FAILED: floor spans $FILES files, expected at least 4"; exit 1; }

# Cardinality is not membership. D14 makes DROPPING a floor member a finding,
# so the eight originals are pinned by NAME and each must still be present —
# without this, one member could vanish while an unrelated failure appeared
# and the count would still pass. Added 2026-08-09 at the arm re-check, which
# found the count-only form certifying exactly that.
MISSING=0
while IFS= read -r want; do
  [ -n "$want" ] || continue
  grep -Fq "$want" "$GROW/only.txt" || { echo "RECEIPT FAILED: floor member dropped -> $want"; MISSING=1; }
done <<'MEMBERS'
admit.test.ts > admitTemplate — effective-config materialization (dimension 3, A5) > materializes the absent previous_reviewer_verdict default and preserves the threshold identity ONCE
admit.test.ts > admitTemplate — producer monopoly (dimension 4, A1: input flags never trusted) > a GATED step with WRONG pre-populated maps → recomputed EXACTLY on the gated rebuild branch too
admit.test.ts > admitTemplate — the gate-free confinement (A8) and own-property write discipline (G8) > a gate-free template admits with a structurally-equal value plus all-false round flags (C38)
fileDefinitionStore.test.ts > ch11-P4 — a gated template loads by ref through the store (file channel → admission) > admits with the effective gate config, the round declaration, and the expanded advancesRound flags
engine.test.ts > the normalizer (ADR-019 D3) — derivation, never validation > writes each registration's EFFECTIVE config into the binding's single config surface
validate.test.ts > ch12-P4 F4 — the defaultAgentConfig roles-entry source form (dimension 3) > defaultActor + defaultAgentConfig together are both delivered
validate.test.ts > dimension 12 — the canonical example round-trip > loads the draft's canonical example to the exact WorkflowTemplate value
validate.test.ts > V10 — roles entries > accepts a present nonempty defaultActor
MEMBERS
[ "$MISSING" = "0" ] || exit 1
echo
echo "M3 RECEIPT OK — growth-only $N in $FILES files (floor 8 in 4, extendable not droppable), baseline-only 0."
