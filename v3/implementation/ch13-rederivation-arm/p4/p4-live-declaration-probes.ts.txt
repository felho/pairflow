const noCatalog = { resolve: () => null };

const base = {
  ref: { id: "probe", version: 1 },
  start: "s1",
  steps: { s1: { role: "r1", instruction: "do", transitions: {} } },
  terminal: ["done"],
  roles: { r1: {} },
};

function run(value: unknown, channel: EngineChannel = { kind: "direct" }) {
  const r = runSurface(templateFormat, value, { channel, catalog: noCatalog });
  return { findings: r.findings.filter((f) => String(f.path).includes("contextBlocks") || String(f.path).includes("ConcernRefs") || String(f.path).includes("capabilityProfile") || f.code !== undefined), ctx: (r.normalized as Record<string, unknown> | undefined)?.["contextBlocks"] };
}

function both(label: string, extra: Record<string, unknown>, yamlExtra: string) {
  const direct = run({ ...base, ...extra });
  const yaml =
    "ref: {id: probe, version: 1}\nstart: s1\nsteps:\n  s1: {role: r1, instruction: do, transitions: {}}\nterminal: [done]\nroles:\n  r1: {}\n" + yamlExtra;
  const doc = parseDocument(yaml);
  const file = run(doc.toJS({ mapAsMap: true }), { kind: "file", doc, source: yaml });
  console.log(label, JSON.stringify({ direct, file, agree: JSON.stringify(direct.findings) === JSON.stringify(file.findings) }));
}
