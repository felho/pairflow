
# L0f runs BEFORE the kernel. It is not a HANDLE/handler: it consumes an operator
# StartCommand + the repo's ProjectConfig and produces a ResolvedStartRequest that
# CREATE_INSTANCE consumes; later START / provider provisioning read the applied values from
# the pinned resolved definition. No instance, no transcript, no version exists yet.

RESOLVE_START(command, project_config) → ResolvedStartRequest | Rejected
  # J1 — definition resolution
  template_id ← command.template_id ?? project_config.default_workflow    # default_workflow is optional convenience
  IF template_id is none THEN RETURN Rejected(no_workflow_selected)        # neither explicit nor defaulted
  source   ← select_source(project_config.workflow_sources, template_id)   # MVP: local(.pairflow/); central store is L11+
  template ← source.load(template_id)                                      # pinned by version
  IF template is none THEN RETURN Rejected(workflow_definition_unavailable)

  # J2 — slot/value resolution
  wf ← project_config.workflows[template_id] ?? empty_workflow_scope      # no per-workflow section → empty scope, not null
  IF command.target is present AND wf.targets[command.target] is none
     THEN RETURN Rejected(unknown_target(command.target))                 # an explicitly-selected target is intentional — a typo rejects (unlike a lenient global slot)
  declared ← { s.id for s in template.slots }
  scope    ← slot_scope(project_config, template_id, command)           # only THIS run's consulted tiers

  # scoped strict reject — unambiguously-scoped tiers only; global defaults stay lenient
  FOR key IN keys(scope.project_wf) ∪ keys(scope.target) ∪ keys(scope.cli):
    IF key NOT IN declared THEN RETURN Rejected(unknown_slot(key))         # typo in a scoped section
  # undeclared keys in shared global defaults — or in OTHER workflows' sections — are ignored for this run

  resolved_slots ← {}
  FOR slot IN template.slots:
    value ← resolve_slot(slot, scope)
    IF value is none AND slot.required THEN RETURN Rejected(unbound_required_slot(slot.id))
    IF value is not none AND NOT type_ok(slot.type, value) THEN RETURN Rejected(slot_type_mismatch(slot.id))
    resolved_slots[slot.id] ← value

  # J3 — variant/mode resolution (start-fixed; the kernel stays mode-agnostic)
  REQUIRE valid_mode_declaration(template)                       # authoring / load-time consistency — fail-closed (see below)
  mode ← command.mode ?? template.default_mode                   # start-fixed; absent ⇒ the template's DECLARED default variant
  IF template.modes is not empty AND mode is none THEN RETURN Rejected(no_mode_selected)   # modes declared but neither command nor default_mode picks one — an explicit authoring choice, never a kernel default
  IF mode is not none AND mode NOT IN template.modes THEN RETURN Rejected(unknown_mode(mode))   # fail-closed: a start mode the template does not declare

  # materialize: slot values, then prune to the mode variant — ONE pinned, immutable definition (J1 template + J2 values + J3 variant).
  # The kernel and L0e read template.runtime_context (etc.) from this PINNED view, exactly as before — no mutable slot map, no runtime mode field.
  resolved_definition ← apply_mode(apply_slots(template, resolved_slots), mode)   # ${slot:…} → values, then prune to the mode variant

  RETURN ResolvedStartRequest {
    resolved_template_ref: pin(resolved_definition),             # pinned RESOLVED definition; CREATE_INSTANCE loads it like any Template
    binding:               project_binding(project_config, command) }   # role → actor: its OWN cascade, not a slot

slot_scope(project_config, template_id, command) → Scope
  wf ← project_config.workflows[template_id] ?? empty_workflow_scope   # no per-workflow section → empty scope, not null
  RETURN Scope {
    project_global: project_config.defaults,                      # global — shared across workflows (lenient)
    project_wf:     wf.defaults,                                  # this workflow's section (strict)
    target:         command.target ? wf.targets[command.target].defaults : { },   # target profile (existence already guarded in RESOLVE_START)
    cli:            command.overrides }                           # explicit CLI/start override (strict)

resolve_slot(slot, scope) → value | none                         # 5-tier cascade; last non-none wins
  value ← slot.default                                            # tier 1: template default
  IF scope.project_global has slot.id THEN value ← scope.project_global[slot.id]   # tier 2: project global
  IF scope.project_wf     has slot.id THEN value ← scope.project_wf[slot.id]       # tier 3: project workflow
  IF scope.target         has slot.id THEN value ← scope.target[slot.id]           # tier 4: target profile
  IF scope.cli            has slot.id THEN value ← scope.cli[slot.id]              # tier 5: CLI/start override (highest)
  RETURN value
          

valid_mode_declaration(template) → ok | Rejected                 # authoring / load-time consistency — fail-closed
  IF template.modes is empty THEN                                # a mono-mode template carries NO mode surface at all
    IF template.default_mode is not none OR any_mode_tag(template)
       THEN RETURN Rejected(mode_surface_without_modes)
    RETURN ok
  IF unsupported_mode_tags(template) is not empty               # a modes: tag OUTSIDE the 2 supported surfaces —
     THEN RETURN Rejected(mode_tag_on_unsupported_surface)      #   a step, an agent_config, or a catalog context_blocks entry
  IF template.default_mode is not none AND template.default_mode NOT IN template.modes
     THEN RETURN Rejected(default_mode_undeclared)               # default_mode ∈ modes
  FOR tag IN allowed_mode_tags(template):                        # gates[].modes / *_refs[].modes
    IF tag NOT IN template.modes THEN RETURN Rejected(undeclared_mode_tag(tag))
  RETURN ok
  # allowed_mode_tags     = modes: tags on the 2 supported surfaces — a gate binding (gates[].modes), a context ref (*_refs[].modes)
  # unsupported_mode_tags = a modes: tag anywhere else (a step, an agent_config, a catalog context_blocks entry); any_mode_tag = either set non-empty

apply_mode(definition, mode) → ResolvedDefinition                # prune to the mode variant at START — NO runtime conditional survives
  IF mode is none THEN RETURN definition                         # mono-mode template — nothing to prune
  # surface 1 — gate-binding presence: in every transition's gate pipeline, keep the bindings whose membership set includes the mode
  #   (a point may declare a code-binding AND a doc-binding — same `uses` / different `config`, or a different `uses`; pruning leaves the mode's)
  FOR step IN definition.steps:
    FOR transition IN step.transitions:
      transition.gates ← [ g for g in transition.gates if in_mode(g, mode) ]
  # surface 2 — context-contribution presence: prune the tagged REFS on EVERY ref-bearing carrier — a role/step
  #   prompt_concern_refs AND a SURVIVING gate's context_block_refs — NEVER the shared catalog context_blocks entry
  #   (a tagged catalog entry would dangle; the catalog stays whole, harmlessly unreferenced in the other mode)
  FOR carrier IN definition.{roles, steps}:
    carrier.prompt_concern_refs ← [ r for r in carrier.prompt_concern_refs if in_mode(r, mode) ]
  FOR step IN definition.steps:
    FOR transition IN step.transitions:
      FOR gate IN transition.gates:                              # the bindings that survived surface 1
        gate.context_block_refs ← [ r for r in gate.context_block_refs if in_mode(r, mode) ]
  RETURN pin(definition)                                         # immutable, mode-specific — the kernel loads it like any Template; no policy map, no runtime mode field

in_mode(tagged, mode) ≜ tagged.modes is empty OR mode IN tagged.modes   # membership predicate; untagged ⇒ present in every mode
