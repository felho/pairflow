
FAIL(instance, reason) → Outcome                                     # kernel_event — any non-terminal

  instance.kernel_status ← TERMINAL ; instance.terminal_disposition ← failed ; instance.failure_reason ← reason   # LC2 a failed run emits the terminal boundary event too — released ONLY if terminal is a declared boundary, post-commit via RECEIVE's hook
