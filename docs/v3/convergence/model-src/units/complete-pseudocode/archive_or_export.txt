
# optional ops (read-only) — a cold snapshot/export; NOT the preservation path
archive_or_export(instance, sink) → ArchiveRef                      # LC4 — copy the durable record + its closure-owned evidence to a cold sink; read-only w.r.t. the run record & runtime (it mutates no canonical run state) but it DOES write catalog metadata (the run_index archive entry below); the system still computes audit/eval from the LIVE durable record / global metrics, never from this copy
  REQUIRE run_index.status(instance.id) ≠ purged                    # input must be NON-PURGED: a purged run has no full record — only the tombstone + global audit floor remain (not a full-run export)
  ref ← cold_copy(canonical_run_record(instance), closure_owned_evidence(instance), sink)
  run_index.upsert(instance.id, status: archived, archive_ref: ref)   # catalog the cold copy in the run_index — a metadata note, NOT preservation; the durable record stays the authority until purge
  RETURN ref
