
# kernel-adjacent — RECEIVE operator_intent dispatch gains DELETE_REQUESTED (alongside CANCEL / SUBMIT_DECISION / RUN_ACTION)
DELETE_REQUESTED(intent) → Outcome                                  # LC4 operator intent — a CANCEL sibling; starts cancel → release → purge. NOT a workflow action.
# The delete chain behind this one intent: a forced release ERRAND (force_release, below) followed by the
# purge contract — hard_purge is a preconditioned storage re-drive, NOT an errand (no request/reply, no completion event)
  instance ← load(intent.instance_id)
  IF instance is none THEN                                          # F-W4-1 resolved (ingress touch): the none path exists — tombstone-before-authority holds STRUCTURALLY here (a purged run has no full run record, hence no binding[operator] to authorize against; only the surviving T1-family tombstone remains to read)
    IF run_index.status(intent.instance_id) = purged THEN RETURN Rejected(already_purged)   # keyed by the INTENT's id — the tombstone survives the record it marks
    RETURN Rejected(unknown_instance)                               # no record and no tombstone ⇒ this run never existed
  IF duplicate(intent.op_id) THEN RETURN Duplicate                  # idempotency-first
  REQUIRE intent.by = instance.binding[operator]                    # operator authority on THIS path (like SUBMIT_DECISION) — BEFORE the purge_pending no-op, so an unauthorized re-delete still gets operator_not_authorized
  IF instance.purge_state = purge_pending THEN RETURN Committed(instance.version, none)   # already pending — a re-issue (fresh op_id) is an idempotent no-op (matrix: no-op once pending / purged)
  IF instance.kernel_status ≠ TERMINAL AND NOT intent.force THEN RETURN Rejected(delete_confirmation_required)   # a LIVE run needs explicit force (v1: --force)
  COMMIT atomically at expected_version = instance.version:
    IF instance.kernel_status ≠ TERMINAL THEN                       # live run ⇒ terminalize first (the lifecycle, unbundled from purge)
      instance.kernel_status ← TERMINAL ; instance.terminal_disposition ← cancelled
    instance.purge_state ← purge_pending                           # MARK only; the destructive purge waits for release (the teeth are on hard_purge, not here)
    instance.transcript.append(DELETE_REQUESTED { op_id: intent.op_id, by: intent.by })
    instance.version ← instance.version + 1
  force_release(instance)                                          # LC4 post-commit operator-FORCED release (below) — NOT the membership-gated boundary hook; the runtime MUST go for purge
  RETURN Committed(instance.version, none)
